class Tomato: #класс Помидора 1
    states = {0: 'отсутствует', 1: 'цветение', 2: 'зеленый', 3: 'красный'} #стадии созревания 2 

    def __init__(self, index): #инициализация 3
        self._index = index
        self._state = 0

    def grow(self): # метод роста томатов 4
        if self._state < 3:
            self._state += 1

    def is_ripe(self): #метод проверки созревания томата 5
        return self._state == 3


class TomatoBush: #класс TomatoBush 6 
    def __init__(self, num_tomatoes):#инициализация 7 
        self.tomatoes = [Tomato(index) for index in range(num_tomatoes)]

    def grow_all(self):#рост всех томатов на кусте 8 
        for tomato in self.tomatoes:
            tomato.grow()

    def all_are_ripe(self): #все ли томаты созрели 9
        return all(tomato.is_ripe() for tomato in self.tomatoes)

    def give_away_all(self): #Сбор урожая 10
        self.tomatoes = []


class Gardener: # класс Gardener 11
    @staticmethod
    def knowledge_base(): # Справка по садоводству 12
        print("Справка по садоводству: ухаживайте за растениями, чтобы они созревали. "
              "Собирайте урожай только, когда все плоды созрели.")

    def __init__(self, name, plant): # инициализация 13
        self.name = name
        self._plant = plant

    def work(self): #Метод ухода за растением 14
        print(f"{self.name} ухаживает за растением.")
        self._plant.grow_all()

    def harvest(self): # Сбор урожая 15
        if self._plant.all_are_ripe():
            print("Все томаты созрели. Садовник собирает урожай.")
            self._plant.give_away_all()
        else:
            print("Томаты еще не созрели! Пожалуйста, продолжайте ухаживать за растением.")


Gardener.knowledge_base() #вызов справки по садоводству 16

bush = TomatoBush(5) #куст с помидорами 17
