import time

def measure_time(func):
    def wrapper(*args, **kwargs):
        start_time = time.time()
        result = func(*args, **kwargs)
        end_time = time.time()
        print(f"Время выполнения функции: {end_time - start_time:.6f} секунд")
        return result
    return wrapper

@measure_time
def fibonacci():
    fib1 = fib2 = 1
    for i in range(2, 200):
        fib1, fib2 = fib2, fib1 + fib2
        print(fib2, end=' ')
    print()

import time


class ExecutionLogger:

    #Декоратор ExecutionLogger используется для логирования начала, окончания и времени выполнения функции.


    def __init__(self, func):
        self.func = func  # Сохраняем ссылку на функцию, к которой применяется декоратор

    def __call__(self, *args, **kwargs):
        # Логируем время начала
        start_time = time.time()
        print(f"Начало выполнения функции '{self.func.__name__}'...")

        # Выполняем саму функцию
        result = self.func(*args, **kwargs)

        # Логируем время окончания и вычисляем длительность
        end_time = time.time()
        duration = end_time - start_time
        print(f"Функция '{self.func.__name__}' завершила выполнение.")
        print(f"Время выполнения: {duration:.4f} секунд.\n")

        return result  # Возвращаем результат выполнения функции


@ExecutionLogger
def calculate_factorial(n):

    ### Функция для вычисления факториала числа n.

    if n == 0:
        return 1
    factorial = 1
    for i in range(1, n + 1):
        factorial *= i
    print(f"Факториал числа {n} равен {factorial}")
    return factorial


@ExecutionLogger
def sum_of_numbers(numbers):

    #Функция для подсчета суммы элементов в списке чисел.

    total_sum = sum(numbers)
    print(f"Сумма чисел {numbers} равна {total_sum}")
    return total_sum


# Тестирование функций с декоратором ExecutionLogger
calculate_factorial(5)
sum_of_numbers([1, 2, 3, 4, 5])


```

import math

class NegativeNumberError(Exception):

   # Исключение NegativeNumberError выбрасывается, если передано отрицательное число, для которого математическая операция не может быть выполнена.


    def __init__(self, value):
        self.value = value
        self.message = f"Ошибка: передано отрицательное число {value}, операция невозможна."
        super().__init__(self.message)


def calculate_square_root(x):

#    Функция для вычисления квадратного корня числа x.

    if x < 0:
        raise NegativeNumberError(x)
    result = math.sqrt(x)
    print(f"Квадратный корень из {x} равен {result}")
    return result


def calculate_logarithm(x):

#    Функция для вычисления натурального логарифма числа x.

    if x <= 0:
        raise NegativeNumberError(x)
    result = math.log(x)
    print(f"Натуральный логарифм числа {x} равен {result}")
    return result


# Тестирование исключений
try:
    calculate_square_root(2)
except NegativeNumberError as e:
    print(e)

try:
    calculate_logarithm(-2)
except NegativeNumberError as e:
    print(e)

```
